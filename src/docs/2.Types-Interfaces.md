/** Mot row trong AG Grid */
export interface GridRow {
  id: string;                          // Unique row ID
  schemaName: string;                  // Display name / schema name
  labels: Record<number, string>;      // LCID -> Label text
  children?: GridRow[];                // Hierarchical data (Form, BPF)
  meta?: Record<string, unknown>;      // Original metadata de dung khi Save
  type?: string;                       // Handler type (dung cho AllInOne)

/** Thay doi cua 1 cell */
export interface CellChange {
  rowId: string;

  oldValue: string;
  newValue: string;

## 2.4 handlers/IHandler.ts

Handler interface — trung tam cua toan bo architecture:

```typescript
import { GridRow, CellChange, LoadResult } from '@/types/grid';
import { AppContextType } from '@/context/AppContext';

/**
 * Moi handler implement interface nay.
 * Handler la PURE — khong goi UI, chi nhan context va tra ve data.
 */
export interface IHandler {
  /** Load data tu Dataverse, tra ve rows cho grid */
  load(context: AppContextType): Promise<LoadResult>;

  /** Save changes nguoc lai Dataverse */
  save(
    rows: GridRow[],
    changes: CellChange[],
    context: AppContextType
  ): Promise<void>;

  /**
   * (Optional) Co cho phep chon component cu the khong?
   * Vi du: FormHandler cho chon form, ContentSnippetHandler cho chon record
   */
  supportsComponentSelection?: boolean;

  /**
   * (Optional) Load danh sach components de user chon
   * Vi du: danh sach forms, danh sach views
   */
  loadComponents?(context: AppContextType): Promise<ComponentOption[]>;
}

/** Option cho component selector */
export interface ComponentOption {
  id: string;
  name: string;
  type?: number;   // Form type, view type, etc.
}
```

## Mapping: Code cu -> Types moi

| Code cu (dynamic) | Type moi |
|---|---|
| `record.w2ui.changes[lcid]` | `CellChange.newValue` |
| `record.recid` | `GridRow.id` |
| `record.schemaName` | `GridRow.schemaName` |
| `record[1033]` (LCID as key) | `GridRow.labels[1033]` |
| `w2ui.grid.records` | `GridRow[]` |
| `window.XrmTranslator.metadata` | Stored in `GridRow.meta` |
| `window.XrmTranslator.*` globals | `AppContextType` |
