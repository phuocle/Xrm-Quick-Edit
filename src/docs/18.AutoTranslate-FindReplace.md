### Translation Modes

Tu code cu `ProposeTranslations()` (dong 401-516):

| Mode | Filter Logic |
|---|---|
| **Missing only** | Target cell is empty |
| **Missing + Identical** | Target is empty OR target == source (chua dich) |
| **Overwrite all** | Translate tat ca rows (ke ca da co value) |

```typescript
function filterRowsForTranslation(
  rows: GridRow[],
  sourceLcid: number,
  targetLcid: number,
  mode: 'missing' | 'missingOrIdentical' | 'overwrite'
): GridRow[] {
  return rows.filter(row => {
    const source = row.labels[sourceLcid];
    const target = row.labels[targetLcid];

    if (!source) return false;  // Khong co source -> skip

    switch (mode) {
      case 'missing': return !target || target.trim() === '';
      case 'missingOrIdentical': return !target || target.trim() === '' || target === source;
      case 'overwrite': return true;
    }
  });
}
```

function applyTranslations(
  results: TranslationResult[],
  rows: GridRow[],
  targetLcid: number
): CellChange[] {
  return results.map(result => {
    const row = rows.find(r => r.labels[sourceLcid] === result.sourceText);
    return {
      rowId: row.id,
      lcid: targetLcid,
      oldValue: row.labels[targetLcid] || '',
      newValue: result.translatedText,
    };
  });
}

### Search Logic

```typescript
function findInGrid(
  rows: GridRow[],
  searchText: string,
  useRegex: boolean,
  targetLcid?: number  // undefined = search tat ca languages
): SearchResult[] {
  const results: SearchResult[] = [];
  const pattern = useRegex ? new RegExp(searchText, 'gi') : null;

  for (const row of flattenAllRows(rows)) {
    const lcids = targetLcid ? [targetLcid] : Object.keys(row.labels).map(Number);

    for (const lcid of lcids) {
      const text = row.labels[lcid];
      if (!text) continue;

      const matches = pattern
        ? pattern.test(text)
        : text.toLowerCase().includes(searchText.toLowerCase());

      if (matches) {
        results.push({
          rowId: row.id,
          lcid,
          currentText: text,
          previewText: pattern
            ? text.replace(pattern, replaceText)
            : text.replace(new RegExp(escapeRegex(searchText), 'gi'), replaceText),
        });
      }
    }
  }

  return results;
}
```

## 18.3 Debug Auto-fill (Development Only)

Tu code cu `ApplyDebugTranslations()` (dong 246-320):

```typescript
// Append "{LCID} {random3chars}" vao source text cho tat ca target languages
// Dung de test UI nhanh ma khong can API key
function debugAutoFill(rows: GridRow[], sourceLcid: number, targetLcids: number[]): CellChange[] {
  const changes: CellChange[] = [];

  for (const row of flattenAllRows(rows)) {
    const source = row.labels[sourceLcid];
    if (!source) continue;

    for (const targetLcid of targetLcids) {
      if (targetLcid === sourceLcid) continue;
      const random = Math.random().toString(36).substring(2, 5);
      changes.push({
        rowId: row.id,
        lcid: targetLcid,
        oldValue: row.labels[targetLcid] || '',
        newValue: `${source} ${targetLcid} ${random}`,
      });
    }
  }

  return changes;
}
```
