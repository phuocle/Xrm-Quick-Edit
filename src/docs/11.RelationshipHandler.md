## 11.3 FillTable — Flat Grid

Tu `FillTable()` (dong 88-125):

```typescript
function buildRelationshipRows(
  relationships: RelationshipMetadata[],
  pluralNames: Map<string, Record<number, string>>
): GridRow[] {
  const rows: GridRow[] = [];

  for (const rel of relationships) {
    const labels: Record<number, string> = {};

    // Lay menu config dua tren loai relationship
    const menuConfig = rel.relType === 'N:N'
      ? rel.Entity1AssociatedMenuConfiguration
      : rel.AssociatedMenuConfiguration;

    const navEntity = getNavEntity(rel); // Entity duoc navigate den
    const entityPluralNames = pluralNames.get(navEntity) || {};

    if (menuConfig.Label?.LocalizedLabels?.length > 0) {
      // Co custom labels
      for (const ll of menuConfig.Label.LocalizedLabels) {
        labels[ll.LanguageCode] = ll.Label;
      }
    } else {
      // Dung plural names lam default
      Object.assign(labels, entityPluralNames);
    }

    rows.push({
      id: rel.MetadataId,
      schemaName: `${rel.SchemaName} (${rel.relType} → ${navEntity})`,
      labels,
      meta: {
        relationship: rel,
        relType: rel.relType,
        navEntity,
        pluralNames: entityPluralNames,
        originalBehavior: menuConfig.Behavior,
      },
    });
  }

  return rows;
}
```

## 11.4 Save Logic

Tu `RelationshipHandler.SaveOnly()` (dong 270-322):

```typescript
async save(rows: GridRow[], changes: CellChange[], context: AppContextType): Promise<void> {
  const changesByRow = groupChangesByRow(changes);

  for (const [rowId, rowChanges] of changesByRow) {
    const row = rows.find(r => r.id === rowId);
    const rel = row.meta.relationship;
    const relType = row.meta.relType;

    // Xay dung menu config
    const menuConfigKey = relType === 'N:N'
      ? 'Entity1AssociatedMenuConfiguration'
      : 'AssociatedMenuConfiguration';

    // Neu chuyen tu UseCollectionName -> UseLabel:
    // Pre-populate labels tu plural names
    const menuLabel = rel[menuConfigKey].Label || { LocalizedLabels: [] };
    if (rel[menuConfigKey].Behavior !== 'UseLabel') {
      // Copy plural names vao labels truoc
      const plurals = row.meta.pluralNames;
      for (const [lcid, name] of Object.entries(plurals)) {
        if (!menuLabel.LocalizedLabels.find(l => l.LanguageCode == lcid)) {
          menuLabel.LocalizedLabels.push({ LanguageCode: parseInt(lcid), Label: name });
        }
      }
    }

    // Apply user changes
    applyChanges(rowChanges, menuLabel.LocalizedLabels);

    // Build update payload
    const odataType = relType === 'N:N'
      ? 'Microsoft.Dynamics.CRM.ManyToManyRelationshipMetadata'
      : 'Microsoft.Dynamics.CRM.OneToManyRelationshipMetadata';

    const payload = {
      '@odata.type': odataType,
      [menuConfigKey]: {
        Behavior: 'UseLabel',  // FORCE UseLabel khi co custom label
        Label: menuLabel,
      },
    };

    // PUT RelationshipDefinitions({id})
    await dataverseService.updateRelationship(rel.MetadataId, payload);
  }

  await publishService.publishEntity(context.selectedEntity);
}
```

### Behavior Switch (QUAN TRONG)

- Khi user edit label, Behavior phai chuyen sang `"UseLabel"`
- Neu giu `"UseCollectionName"`, CRM se ignore custom labels
- Code cu xu ly viec nay bang cach pre-populate plural names roi apply changes

## 11.5 Diem Khac Biet 1:N vs N:N

| | 1:N / N:1 | N:N |
|---|---|---|
| Menu config | `AssociatedMenuConfiguration` | `Entity1AssociatedMenuConfiguration` |
| OData type | `OneToManyRelationshipMetadata` | `ManyToManyRelationshipMetadata` |
| Nav entity | ReferencingEntity (1:N) hoac ReferencedEntity (N:1) | Entity2LogicalName |
