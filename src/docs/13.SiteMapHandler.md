## 13.3 Parse SiteMap XML

```typescript
interface SiteMapNode {
  id: string;         // composite: areaId|groupId|subAreaId
  nodeType: 'Area' | 'Group' | 'SubArea';
  nodeId: string;     // Id attribute
  entity?: string;    // SubArea entity (optional)
  labels: Record<number, string>;
}

function parseSiteMapXml(doc: Document, component: string): GridRow[] {
  const rows: GridRow[] = [];
  const tagName = component === 'DisplayName' ? 'Title' : 'Description';

  const areas = doc.querySelectorAll('Area');
  for (const area of areas) {
    const areaId = area.getAttribute('Id')!;
    rows.push(buildNodeRow(area, areaId, 'Area', tagName));

    const groups = area.querySelectorAll(':scope > Group');
    for (const group of groups) {
      const groupId = group.getAttribute('Id')!;
      const compositeId = `${areaId}|${groupId}`;
      rows.push(buildNodeRow(group, compositeId, 'Group', tagName));

      const subAreas = group.querySelectorAll(':scope > SubArea');
      for (const subArea of subAreas) {
        const subAreaId = subArea.getAttribute('Id')!;
        const entity = subArea.getAttribute('Entity');
        const compositeId3 = `${areaId}|${groupId}|${subAreaId}`;
        rows.push(buildNodeRow(subArea, compositeId3, 'SubArea', tagName, entity));
      }
    }
  }

  return rows;
}

function buildNodeRow(
  node: Element, compositeId: string, nodeType: string, tagName: string, entity?: string
): GridRow {
  const labels: Record<number, string> = {};
  const titleElements = node.querySelectorAll(`:scope > ${tagName}`);

  for (const el of titleElements) {
    const lcid = parseInt(el.getAttribute('LCID')!);
    const text = el.getAttribute(tagName)!;
    labels[lcid] = text;
  }

  return {
    id: compositeId,
    schemaName: `[${nodeType}] ${node.getAttribute('Id')}${entity ? ` (${entity})` : ''}`,
    labels,
    meta: { nodeType, nodeId: node.getAttribute('Id'), entity },
  };
}
```

## 13.5 Save — XML DOM Update

Tu `Save()` (dong 387-482):

```typescript
async save(rows: GridRow[], changes: CellChange[], context: AppContextType): Promise<void> {
  // 1. Parse sitemapxml lai
  const parser = new DOMParser();
  const doc = parser.parseFromString(currentSiteMapXml, 'text/xml');

  // 2. Apply changes
  const tagName = context.selectedComponent === 'DisplayName' ? 'Title' : 'Description';

  for (const change of changes) {
    const row = rows.find(r => r.id === change.rowId);
    const node = findNodeByCompositeId(doc, row.meta);

    if (!node) continue;

    // Tim Title/Description element cho language nay
    const existing = node.querySelector(
      `:scope > ${tagName}[LCID="${change.lcid}"]`
    );

    if (existing) {
      existing.setAttribute(tagName, change.newValue);
    } else {
      // Tao moi
      const newEl = doc.createElement(tagName);
      newEl.setAttribute('LCID', String(change.lcid));
      newEl.setAttribute(tagName, change.newValue);
      node.appendChild(newEl);
    }
  }

  // 3. Serialize lai XML
  const serializer = new XMLSerializer();
  const updatedXml = serializer.serializeToString(doc);

  // 4. Update sitemap record
  await dataverseService.updateRecord('sitemaps', sitemapId, {
    sitemapxml: updatedXml,
  });

  // 5. Publish — KHAC BIET: PublishAllXml (khong phai entity publish)
  // Phai RunAsBaseLanguage
  await context.runAsBaseLanguage(async () => {
    await dataverseAPI.execute({
      operationName: 'PublishAllXml',
      operationType: 0,
    });
  });
}
```

> **QUAN TRONG**: SiteMap publish khong dung `PublishXml` voi entity name.
> Phai dung `PublishAllXml` va wrap trong `RunAsBaseLanguage()`.
