## 10.1 Tong Quan

OptionSet la mot trong nhung handler phuc tap nhat vi:
- **5 loai** option set metadata khac nhau (Picklist, Boolean, Status, State, MultiSelect)
- **Compound ID**: `{attributeId}|{optionValue}` vi nhieu attributes co the share option value
- **Global vs Local**: Payload va publish logic khac nhau
- **Hierarchical grid**: Parent = attribute name, children = option values

## 10.3 FillTable — Hierarchical Grid

Tu `FillTable()` (dong 183-210) va `HandleOptionSets()` (dong 149-181):

```typescript
function buildOptionSetRows(attributes: AttributeMetadata[], component: string): GridRow[] {
  const rows: GridRow[] = [];
  const compField = component === 'DisplayName' ? 'Label' : component;

  for (const attr of attributes) {
    const optionSet = attr.OptionSet || attr.GlobalOptionSet;
    if (!optionSet) continue;

    // Boolean option sets co TrueOption/FalseOption thay vi Options array
    const options = optionSet.TrueOption
      ? [optionSet.TrueOption, optionSet.FalseOption]
      : optionSet.Options;

    if (!options || options.length === 0) continue;

    // Parent row (attribute name) — khong edit duoc
    const parent: GridRow = {
      id: attr.MetadataId,
      schemaName: attr.LogicalName,
      labels: {},
      children: [],
      meta: { attribute: attr, isGlobal: optionSet.IsGlobal },
    };

    // Children (option values) — edit duoc
    for (const option of options) {
      const labels: Record<number, string> = {};
      const localizedLabels = option[compField]?.LocalizedLabels || [];

      for (const ll of localizedLabels) {
        labels[ll.LanguageCode] = ll.Label;
      }

      parent.children!.push({
        id: `${attr.MetadataId}|${option.Value}`,  // COMPOUND ID
        schemaName: String(option.Value),
        labels,
        meta: { optionValue: option.Value },
      });
    }

    rows.push(parent);
  }

  return rows;
}
```

> **Compound ID pattern**: `"{attributeMetadataId}|{optionValue}"`.
> Khi save, parse lai: `id.split('|')` -> `[attributeId, optionValue]`.

## 10.5 Save — UpdateOptionValue

Tu `OptionSetHandler.SaveOnly()` (dong 258-280):

```typescript
async save(rows: GridRow[], changes: CellChange[], context: AppContextType): Promise<void> {
  const updates: UpdateOptionValuePayload[] = [];

  for (const change of changes) {
    // Parse compound ID
    const [attributeId, optionValueStr] = change.rowId.split('|');
    const optionValue = parseInt(optionValueStr);
    const parentRow = rows.find(r => r.id === attributeId);
    const attr = parentRow.meta.attribute;
    const isGlobal = parentRow.meta.isGlobal;

    // Build payload
    const payload: any = {
      Value: optionValue,
      [getOptionSetComponent(context.selectedComponent)]: {
        LocalizedLabels: [{ LanguageCode: change.lcid, Label: change.newValue }],
      },
      MergeLabels: true,
    };

    if (isGlobal) {
      // Global option set: dung OptionSetName
      payload.OptionSetName = attr.GlobalOptionSet.Name;
    } else {
      // Local option set: dung EntityLogicalName + AttributeLogicalName
      payload.EntityLogicalName = context.selectedEntity.toLowerCase();
      payload.AttributeLogicalName = attr.LogicalName;
    }

    updates.push(payload);
  }

  // POST UpdateOptionValue cho tung update
  for (const payload of updates) {
    await dataverseService.updateOptionValue(payload);
  }

  // Publish: entity + global option sets rieng
  await publishService.publishEntity(context.selectedEntity);
  // Neu co global option sets, publish rieng
  const globalNames = getUniqueGlobalOptionSetNames(updates);
  if (globalNames.length > 0) {
    await publishService.publishGlobalOptionSets(globalNames);
  }
}
```
