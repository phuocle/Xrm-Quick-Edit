## 19.3 AttributePropertyHandler — Load

Tu `AttributePropertyHandler.js` (dong ~70-130):

```typescript
async load(context: AppContextType): Promise<PropertyLoadResult> {
  const entityId = context.entityMetadataId;

  // GET EntityDefinitions({id})/Attributes?$filter=IsCustomizable/Value eq true
  const attributes = await dataverseService.getAttributes(context.selectedEntity);

  const rows: PropertyRow[] = attributes
    .sort((a, b) => a.SchemaName.localeCompare(b.SchemaName))
    .map(attr => ({
      id: attr.MetadataId,
      schemaName: attr.SchemaName,
      attributeType: attr.AttributeType,
      requiredLevel: attr.RequiredLevel?.Value || 'None',
      isAuditEnabled: attr.IsAuditEnabled?.Value ?? false,
      isValidForAdvancedFind: attr.IsValidForAdvancedFind?.Value ?? false,
      isSecured: attr.IsSecured ?? false,
      meta: { attribute: attr },
    }));

  return { rows };
}
```

## 19.5 Save

Tu `AttributePropertyHandler.js` (dong ~130-190):

```typescript
async save(rows: PropertyRow[], changes: PropertyChange[]): Promise<void> {
  for (const change of changes) {
    const row = rows.find(r => r.id === change.rowId);
    const attr = row.meta.attribute;

    // Build update payload
    const updatePayload: Record<string, any> = {};

    if (change.field === 'requiredLevel') {
      // EDGE CASE: Khong cho phep doi TU SystemRequired
      if (attr.RequiredLevel?.Value === 'SystemRequired') {
        throw new Error(`Cannot change SystemRequired field: ${attr.SchemaName}`);
      }
      updatePayload.RequiredLevel = { Value: change.newValue };
    }
    else if (change.field === 'isAuditEnabled') {
      updatePayload.IsAuditEnabled = { Value: change.newValue };
    }
    else if (change.field === 'isValidForAdvancedFind') {
      updatePayload.IsValidForAdvancedFind = { Value: change.newValue };
    }
    else if (change.field === 'isSecured') {
      updatePayload.IsSecured = change.newValue;
    }

    // PUT voi MergeLabels: true
    await dataverseService.updateAttribute(
      context.selectedEntity,
      attr.MetadataId,
      { ...attr, ...updatePayload }
    );
  }

  // Publish
  await publishService.publishEntity(context.selectedEntity);
}
```

## 19.6 Edge Cases

| Edge Case | Xu Ly |
|---|---|
| SystemRequired fields | Khong cho phep doi tu SystemRequired sang value khac |
| IsSecured | Chi available neu entity co field-level security enabled |
| Audit | Chi co tac dung neu entity-level audit enabled |
| Bulk edit | Nhieu rows co the thay doi cung luc |
