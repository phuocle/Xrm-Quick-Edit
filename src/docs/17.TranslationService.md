## 17.2 Provider Interface

```typescript
// services/translationService.ts

export interface TranslationProvider {
  id: TranslationProviderId;
  name: string;
  requiresApiKey: boolean;
  translate(request: TranslationRequest): Promise<TranslationResult[]>;
}

export interface TranslationRequest {
  texts: string[];
  sourceLang: string;   // ISO 639-1: "en", "fr", "vi"
  targetLang: string;
  config?: TranslationConfig;
}

export interface TranslationResult {
  sourceText: string;
  translatedText: string;
}

export interface TranslationConfig {
  apiKey: string;
  model?: string;         // Gemini model name
  customPrompt?: string;  // Gemini custom prompt
}
```

## 17.3 Glosbe Provider (Free)

Tu code cu (dong ~680-750):

```typescript
const glosbeProvider: TranslationProvider = {
  id: 'glosbe',
  name: 'Glosbe (Free)',
  requiresApiKey: false,

  async translate(request: TranslationRequest): Promise<TranslationResult[]> {
    const results: TranslationResult[] = [];

    // Glosbe: 1 request per word/phrase (khong co batch)
    for (const text of request.texts) {
      const url = `https://glosbe.com/gapi/translate?from=${request.sourceLang}&dest=${request.targetLang}&phrase=${encodeURIComponent(text)}&format=json`;

      const response = await fetch(url);
      const data = await response.json();

      const translation = data.tuc?.[0]?.phrase?.text || text;
      results.push({ sourceText: text, translatedText: translation });
    }

    return results;
  },
};
```

> **Luu y**: Glosbe co rate limiting va chat luong thap. Chi dung cho demo/fallback.

## 17.4 Azure Translator Provider

Tu code cu (dong ~600-680):

```typescript
const azureProvider: TranslationProvider = {
  id: 'azure',
  name: 'Azure Translator',
  requiresApiKey: true,

  async translate(request: TranslationRequest): Promise<TranslationResult[]> {
    const url = `https://api.cognitive.microsofttranslator.com/translate?api-version=3.0&from=${request.sourceLang}&to=${request.targetLang}`;

    const body = request.texts.map(text => ({ Text: text }));

    const response = await fetch(url, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Ocp-Apim-Subscription-Key': request.config!.apiKey,
      },
      body: JSON.stringify(body),
    });

    const data = await response.json();

    return data.map((item: any, i: number) => ({
      sourceText: request.texts[i],
      translatedText: item.translations[0].text,
    }));
  },
};
```

## 17.5 DeepL Provider

Tu code cu (dong ~550-600):

```typescript
const deeplProvider: TranslationProvider = {
  id: 'deepl',
  name: 'DeepL',
  requiresApiKey: true,

  async translate(request: TranslationRequest): Promise<TranslationResult[]> {
    // DeepL dung api-free.deepl.com (free) hoac api.deepl.com (pro)
    const url = 'https://api-free.deepl.com/v2/translate';

    const params = new URLSearchParams();
    params.append('source_lang', request.sourceLang.toUpperCase());
    params.append('target_lang', request.targetLang.toUpperCase());
    for (const text of request.texts) {
      params.append('text', text);
    }

    const response = await fetch(url, {
      method: 'POST',
      headers: {
        'Authorization': `DeepL-Auth-Key ${request.config!.apiKey}`,
        'Content-Type': 'application/x-www-form-urlencoded',
      },
      body: params.toString(),
    });

    const data = await response.json();

    return data.translations.map((t: any, i: number) => ({
      sourceText: request.texts[i],
      translatedText: t.text,
    }));
  },
};
```

## 17.6 Gemini AI Provider (QUAN TRONG NHAT)

Tu code cu (dong 111-194) — phuc tap nhat vi batch translation voi custom prompt:

```typescript
const geminiProvider: TranslationProvider = {
  id: 'gemini',
  name: 'Gemini AI',
  requiresApiKey: true,

  async translate(request: TranslationRequest): Promise<TranslationResult[]> {
    const model = request.config?.model || 'gemini-2.0-flash';
    const url = `https://generativelanguage.googleapis.com/v1beta/models/${model}:generateContent?key=${request.config!.apiKey}`;

    // System instruction
    const systemPrompt = request.config?.customPrompt ||
      `You are a professional translator. Translate the following phrases from ${request.sourceLang} to ${request.targetLang}. Return ONLY a JSON array of translated strings, maintaining the exact same order and count.`;

    const body = {
      system_instruction: {
        parts: [{ text: systemPrompt }],
      },
      contents: [{
        parts: [{
          text: JSON.stringify(request.texts),
        }],
      }],
    };

    const response = await fetch(url, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(body),
    });

    const data = await response.json();

    // Parse response — can xu ly markdown fences
    let responseText = data.candidates[0].content.parts[0].text;

    // Strip markdown code fences neu co
    responseText = responseText.replace(/```json\n?/g, '').replace(/```\n?/g, '').trim();

    const translations: string[] = JSON.parse(responseText);

    // Validate: so luong phai khop
    if (translations.length !== request.texts.length) {
      throw new Error(
        `Gemini returned ${translations.length} translations but expected ${request.texts.length}`
      );
    }

    return translations.map((text, i) => ({
      sourceText: request.texts[i],
      translatedText: text,
    }));
  },
};
```

> **Edge case**: Gemini co the tra ve markdown fences (`\`\`\`json ... \`\`\``) quanh JSON array.
> Phai strip truoc khi parse.

async function getGeminiConfig(): Promise<{ apiKey: string; model: string; customPrompt: string }> {
  const config = await toolboxAPI.settings.get('geminiConfig');
